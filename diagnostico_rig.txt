import * as THREE from 'three';
import { RIG_PRESET_DEFINITIONS, RigPreset } from '../types/studioConfig';
import { ImportedUvMesh } from '../components/Viewport3D';

function distToSegmentSq(
  p: THREE.Vector3,
  a: THREE.Vector3,
  b: THREE.Vector3,
  scratchAb: THREE.Vector3,
  scratchAp: THREE.Vector3
): number {
  scratchAb.subVectors(b, a);
  scratchAp.subVectors(p, a);
  const abLenSq = scratchAb.lengthSq();
  if (abLenSq < 1e-8) return scratchAp.lengthSq();
  const t = THREE.MathUtils.clamp(scratchAp.dot(scratchAb) / abLenSq, 0, 1);
  scratchAb.multiplyScalar(t).add(a);
  return p.distanceToSquared(scratchAb);
}

/**
 * Generates clean box-projected UV0 coordinates in [0, 1] for any meshes inside `group`
 * (or repairs missing UVs) and returns extracted UV edge arrays for the 2D UV Inspector.
 */
export function autoGenerateUvsForGroup(group: THREE.Object3D): ImportedUvMesh[] {
  group.updateMatrixWorld(true);
  const results: ImportedUvMesh[] = [];
  const meshes: THREE.Mesh[] = [];

  group.traverse((obj) => {
    if ((obj as THREE.Mesh).isMesh) {
      const m = obj as THREE.Mesh;
      // Skip outline helper meshes
      if (m.material instanceof THREE.MeshBasicMaterial && m.material.side === THREE.BackSide && !m.name) {
        return;
      }
      meshes.push(m);
    }
  });

  if (meshes.length === 0) return results;

  // Pack all meshes into an NxN grid of UV tiles inside [0.02, 0.98] so islands never overlap
  const gridCols = Math.ceil(Math.sqrt(meshes.length));
  const cellSpan = 1 / gridCols;
  const pad = cellSpan * 0.06;
  const innerSpan = Math.max(0.02, cellSpan - pad * 2);

  const pA = new THREE.Vector3();
  const pB = new THREE.Vector3();
  const pC = new THREE.Vector3();
  const e1 = new THREE.Vector3();
  const e2 = new THREE.Vector3();
  const fn = new THREE.Vector3();

  meshes.forEach((mesh, meshIdx) => {
    const geo = mesh.geometry;
    if (!geo || !geo.attributes.position) return;

    // Convert to non-indexed if indexed so triplanar face projection has zero seam stretching
    const workingGeo = geo.index ? geo.toNonIndexed() : geo;
    if (workingGeo !== geo) {
      mesh.geometry = workingGeo;
      geo.dispose();
    }

    const pos = workingGeo.attributes.position;
    workingGeo.computeBoundingBox();
    const bbox = workingGeo.boundingBox!;
    const size = new THREE.Vector3();
    bbox.getSize(size);
    const maxDim = Math.max(size.x, size.y, size.z, 1e-4);

    const uvArray = new Float32Array(pos.count * 2);
    const col = meshIdx % gridCols;
    const row = Math.floor(meshIdx / gridCols);
    const tileX = col * cellSpan + pad;
    const tileY = row * cellSpan + pad;

    const triCount = Math.floor(pos.count / 3);
    for (let t = 0; t < triCount; t++) {
      const i0 = t * 3;
      const i1 = t * 3 + 1;
      const i2 = t * 3 + 2;

      pA.fromBufferAttribute(pos, i0);
      pB.fromBufferAttribute(pos, i1);
      pC.fromBufferAttribute(pos, i2);

      e1.subVectors(pB, pA);
      e2.subVectors(pC, pA);
      fn.crossVectors(e1, e2);

      const ax = Math.abs(fn.x);
      const ay = Math.abs(fn.y);
      const az = Math.abs(fn.z);

      const verts = [pA, pB, pC];
      for (let k = 0; k < 3; k++) {
        const v = verts[k];
        let uLocal = 0;
        let vLocal = 0;
        if (ax >= ay && ax >= az) {
          uLocal = (v.z - bbox.min.z) / maxDim;
          vLocal = (v.y - bbox.min.y) / maxDim;
        } else if (ay >= ax && ay >= az) {
          uLocal = (v.x - bbox.min.x) / maxDim;
          vLocal = (v.z - bbox.min.z) / maxDim;
        } else {
          uLocal = (v.x - bbox.min.x) / maxDim;
          vLocal = (v.y - bbox.min.y) / maxDim;
        }
        const vi = (t * 3 + k) * 2;
        uvArray[vi] = tileX + THREE.MathUtils.clamp(uLocal, 0, 1) * innerSpan;
        uvArray[vi + 1] = tileY + THREE.MathUtils.clamp(vLocal, 0, 1) * innerSpan;
      }
    }

    const uvAttr = new THREE.BufferAttribute(uvArray, 2);
    workingGeo.setAttribute('uv', uvAttr);
    workingGeo.setAttribute('uv1', uvAttr);
    workingGeo.setAttribute('uv2', uvAttr);
    workingGeo.computeVertexNormals();

    const maxEdgesTris = Math.min(triCount, 12000);
    const edges = new Float32Array(maxEdgesTris * 12);
    for (let t = 0; t < maxEdgesTris; t++) {
      const i0 = t * 3;
      const i1 = t * 3 + 1;
      const i2 = t * 3 + 2;
      const o = t * 12;
      edges[o] = uvAttr.getX(i0);
      edges[o + 1] = uvAttr.getY(i0);
      edges[o + 2] = uvAttr.getX(i1);
      edges[o + 3] = uvAttr.getY(i1);
      edges[o + 4] = uvAttr.getX(i1);
      edges[o + 5] = uvAttr.getY(i1);
      edges[o + 6] = uvAttr.getX(i2);
      edges[o + 7] = uvAttr.getY(i2);
      edges[o + 8] = uvAttr.getX(i2);
      edges[o + 9] = uvAttr.getY(i2);
      edges[o + 10] = uvAttr.getX(i0);
      edges[o + 11] = uvAttr.getY(i0);
    }

    results.push({
      name: mesh.name || `Malha_${meshIdx + 1}`,
      uvEdges: edges,
      triCount,
    });
  });

  return results;
}

export interface AutoRigResult {
  clips: THREE.AnimationClip[];
  boneCount: number;
  armatureRoot: THREE.Group;
  skinnedMeshes: THREE.SkinnedMesh[];

/**
 * Automatically builds a real Three.js Skeleton hierarchy matching `preset` fitted to `group`,
 * computes inverse-distance vertex weights, converts meshes to SkinnedMeshes, and generates
 * playable AnimationClips compatible with GLTFExporter.
 */
export function autoRigAndAnimateGroup(group: THREE.Group, preset: RigPreset): AutoRigResult {
  const prevArmature = group.getObjectByName('AutoRig_Armature');
  if (prevArmature) {
    group.remove(prevArmature);
  }

  group.updateMatrixWorld(true);
  const groupInvWorld = group.matrixWorld.clone().invert();

  // Compute bounding box in group-local space
  const box = new THREE.Box3();
  const vTmp = new THREE.Vector3();
  const targetMeshes: THREE.Mesh[] = [];

  group.traverse((obj) => {
    if ((obj as THREE.Mesh).isMesh) {
      const m = obj as THREE.Mesh;
      if (m.material instanceof THREE.MeshBasicMaterial && m.material.side === THREE.BackSide && !m.name) {
        return;
      }
      targetMeshes.push(m);
      const pos = m.geometry?.attributes?.position;
      if (pos) {
        const meshToGroup = new THREE.Matrix4().multiplyMatrices(groupInvWorld, m.matrixWorld);
        for (let i = 0; i < pos.count; i++) {
          vTmp.fromBufferAttribute(pos, i).applyMatrix4(meshToGroup);
          box.expandByPoint(vTmp);
        }
      }
    }
  });

  if (box.isEmpty()) {
    box.set(new THREE.Vector3(-0.4, 0, -0.4), new THREE.Vector3(0.4, 1.75, 0.4));
  }

  const size = new THREE.Vector3();
  const center = new THREE.Vector3();
  box.getSize(size);
  box.getCenter(center);

  const def = RIG_PRESET_DEFINITIONS[preset] || RIG_PRESET_DEFINITIONS.HUMANOIDE;

  // Determine normalized preset bounds in Three.js Y-up space
  const refBox = new THREE.Box3();
  const headPositions = new Map<string, THREE.Vector3>();
  const tailPositions = new Map<string, THREE.Vector3>();

  for (const [bName, , headB, tailB] of def.bones) {
    const h = new THREE.Vector3(headB[0], headB[2], -headB[1]);
    const t = new THREE.Vector3(tailB[0], tailB[2], -tailB[1]);
    refBox.expandByPoint(h);
    refBox.expandByPoint(t);
    headPositions.set(bName, h);
    tailPositions.set(bName, t);
  }

  const refSize = new THREE.Vector3();
  const refCenter = new THREE.Vector3();
  refBox.getSize(refSize);
  refBox.getCenter(refCenter);

  const scaleX = (size.x || 1) / Math.max(refSize.x, 0.15);
  const scaleY = (size.y || 1) / Math.max(refSize.y, 0.25);
  const scaleZ = (size.z || 1) / Math.max(refSize.z, 0.15);
  const uniformFootprintScale = Math.min(scaleX, scaleZ);

  const mapToGroupSpace = (p: THREE.Vector3) =>
    new THREE.Vector3(
      center.x + (p.x - refCenter.x) * uniformFootprintScale,
      box.min.y + (p.y - refBox.min.y) * scaleY,
      center.z + (p.z - refCenter.z) * uniformFootprintScale
    );

  // Build THREE.Bone hierarchy in group-local space
  const armatureRoot = new THREE.Group();
  armatureRoot.name = 'AutoRig_Armature';
  group.add(armatureRoot);

  const bones: THREE.Bone[] = [];
  const bonesByName = new Map<string, THREE.Bone>();
  const worldSegments: { index: number; head: THREE.Vector3; tail: THREE.Vector3 }[] = [];

  for (const [bName, parentName] of def.bones) {
    const hGroup = mapToGroupSpace(headPositions.get(bName)!);
    const tGroup = mapToGroupSpace(tailPositions.get(bName)!);

    const bone = new THREE.Bone();
    bone.name = bName;

    const parentBone = parentName ? bonesByName.get(parentName) : undefined;
    if (parentBone) {
      const parentHGroup = mapToGroupSpace(headPositions.get(parentName!)!);
      bone.position.copy(hGroup).sub(parentHGroup);
      parentBone.add(bone);
    } else {
      bone.position.copy(hGroup);
      armatureRoot.add(bone);
    }

    worldSegments.push({ index: bones.length, head: hGroup, tail: tGroup });
    bones.push(bone);
    bonesByName.set(bName, bone);
  }

  group.updateMatrixWorld(true);
  const skeleton = new THREE.Skeleton(bones);
  skeleton.calculateInverses();

  const skinnedMeshes: THREE.SkinnedMesh[] = [];
  const scratchAb = new THREE.Vector3();
  const scratchAp = new THREE.Vector3();

  for (const oldMesh of targetMeshes) {
    const geo = oldMesh.geometry;
    if (!geo || !geo.attributes.position) continue;

    // Remove any rigid outline helper child
    const outlineChildren = oldMesh.children.filter(
      (c) => (c as THREE.Mesh).isMesh && (c as THREE.Mesh).material instanceof THREE.MeshBasicMaterial
    );
    outlineChildren.forEach((c) => oldMesh.remove(c));

    const posAttr = geo.attributes.position;
    const vertCount = posAttr.count;
    const skinIndices = new Uint16Array(vertCount * 4);
    const skinWeights = new Float32Array(vertCount * 4);

    const meshToGroup = new THREE.Matrix4().multiplyMatrices(groupInvWorld, oldMesh.matrixWorld);
    const bestIdx = new Int32Array(4);
    const bestW = new Float32Array(4);

    for (let i = 0; i < vertCount; i++) {
      vTmp.fromBufferAttribute(posAttr, i).applyMatrix4(meshToGroup);
      bestIdx.fill(0);
      bestW.fill(0);

      for (let b = 0; b < worldSegments.length; b++) {
        const seg = worldSegments[b];
        const dSq = distToSegmentSq(vTmp, seg.head, seg.tail, scratchAb, scratchAp);
        const w = 1 / Math.pow(dSq + 0.0025, 2.0);

        // Insert into top 4
        if (w > bestW[3]) {
          let pos = 3;
          while (pos > 0 && w > bestW[pos - 1]) {
            bestW[pos] = bestW[pos - 1];
            bestIdx[pos] = bestIdx[pos - 1];
            pos--;
          }
          bestW[pos] = w;
          bestIdx[pos] = seg.index;
        }
      }

      const sum = bestW[0] + bestW[1] + bestW[2] + bestW[3] || 1;
      const o = i * 4;
      for (let k = 0; k < 4; k++) {
        skinIndices[o + k] = bestIdx[k];
        skinWeights[o + k] = bestW[k] / sum;
      }
    }

    geo.setAttribute('skinIndex', new THREE.Uint16BufferAttribute(skinIndices, 4));
    geo.setAttribute('skinWeight', new THREE.Float32BufferAttribute(skinWeights, 4));

    let skinned: THREE.SkinnedMesh;
    if ((oldMesh as THREE.SkinnedMesh).isSkinnedMesh) {
      skinned = oldMesh as THREE.SkinnedMesh;
    } else {
      skinned = new THREE.SkinnedMesh(geo, oldMesh.material);
      skinned.name = oldMesh.name;
      skinned.position.copy(oldMesh.position);
      skinned.quaternion.copy(oldMesh.quaternion);
      skinned.scale.copy(oldMesh.scale);
      skinned.castShadow = oldMesh.castShadow;
      skinned.receiveShadow = oldMesh.receiveShadow;

      const parent = oldMesh.parent;
      if (parent) {
        parent.add(skinned);
        parent.remove(oldMesh);
      }
    }

    skinned.updateMatrixWorld(true);
    skinned.bind(skeleton, skinned.matrixWorld.clone());
    skinnedMeshes.push(skinned);
  }

  const clips = buildClipsForPreset(preset, bonesByName);
  return {
    clips,
    boneCount: bones.length,
    armatureRoot,
    skinnedMeshes,
  };
}

function createStarShape(points = 5, outerR = 0.42, innerR = 0.19): THREE.Shape {
  const shape = new THREE.Shape();
  for (let i = 0; i < points * 2; i++) {
    const r = i % 2 === 0 ? outerR : innerR;
    const angle = (i / (points * 2)) * Math.PI * 2 - Math.PI / 2;
    const x = Math.cos(angle) * r;
    const y = Math.sin(angle) * r;
    if (i === 0) shape.moveTo(x, y);
    else shape.lineTo(x, y);
  }
  shape.closePath();
  return shape;
}

function createPolygonShape(sides: number, radius: number): THREE.Shape {
  const shape = new THREE.Shape();
  for (let i = 0; i < sides; i++) {
    const a = (i / sides) * Math.PI * 2;
    const x = Math.cos(a) * radius;
    const y = Math.sin(a) * radius;
    if (i === 0) shape.moveTo(x, y);
    else shape.lineTo(x, y);
  }
  shape.closePath();
  return shape;
}

function createRingBevelShape(outerR = 0.4, innerR = 0.2): THREE.Shape {
  const shape = new THREE.Shape();
  shape.absarc(0, 0, outerR, 0, Math.PI * 2, false);
  const hole = new THREE.Path();
  hole.absarc(0, 0, innerR, 0, Math.PI * 2, true);
  shape.holes.push(hole);
  return shape;
}

function createHeartShape(): THREE.Shape {
  const shape = new THREE.Shape();
  const s = 0.022;
  shape.moveTo(0, 5 * s);
  shape.bezierCurveTo(0, 5 * s, -2 * s, 12 * s, -10 * s, 12 * s);
  shape.bezierCurveTo(-18 * s, 12 * s, -18 * s, 1 * s, -18 * s, 1 * s);
  shape.bezierCurveTo(-18 * s, -7 * s, -10 * s, -13 * s, 0, -19 * s);
  shape.bezierCurveTo(10 * s, -13 * s, 18 * s, -7 * s, 18 * s, 1 * s);
  shape.bezierCurveTo(18 * s, 1 * s, 18 * s, 12 * s, 10 * s, 12 * s);
  shape.bezierCurveTo(2 * s, 12 * s, 0, 5 * s, 0, 5 * s);
  return shape;
}

export function createSplinePrimitiveGeometry(kind: PrimitiveShapeKind): THREE.BufferGeometry {
  switch (kind) {
    case 'cube':
      return new THREE.BoxGeometry(0.65, 0.65, 0.65, 2, 2, 2);
    case 'sphere':
      return new THREE.SphereGeometry(0.38, 32, 22);
    case 'torus':
      return new THREE.TorusGeometry(0.34, 0.13, 20, 44);
    case 'cylinder':
      return new THREE.CylinderGeometry(0.28, 0.28, 0.72, 28);
    case 'cone':
      return new THREE.ConeGeometry(0.34, 0.74, 28);
    case 'capsule':
      return new THREE.CapsuleGeometry(0.22, 0.42, 12, 24);
    case 'icosahedron':
      return new THREE.IcosahedronGeometry(0.4, 0);
    case 'torus_knot':
      return new THREE.TorusKnotGeometry(0.28, 0.095, 96, 16);
    case 'star': {
      const geo = new THREE.ExtrudeGeometry(createStarShape(), {
        depth: 0.16,
        bevelEnabled: true,
        bevelThickness: 0.04,
        bevelSize: 0.03,
        bevelSegments: 3,
      });
      geo.center();
      return geo;
    }
    case 'hex': {
      const geo = new THREE.ExtrudeGeometry(createPolygonShape(6, 0.38), {
        depth: 0.2,
        bevelEnabled: true,
        bevelThickness: 0.045,
        bevelSize: 0.035,
        bevelSegments: 3,
      });
      geo.center();
      return geo;
    }
    case 'ring_bevel': {
      const geo = new THREE.ExtrudeGeometry(createRingBevelShape(), {
        depth: 0.14,
        bevelEnabled: true,
        bevelThickness: 0.04,
        bevelSize: 0.03,
        bevelSegments: 3,
        curveSegments: 32,
      });
      geo.center();
      return geo;
    }
    case 'heart': {
      const geo = new THREE.ExtrudeGeometry(createHeartShape(), {
        depth: 0.14,
        bevelEnabled: true,
        bevelThickness: 0.05,
        bevelSize: 0.04,
        bevelSegments: 4,
      });
      geo.center();
      return geo;
    }
  }
}

export function getStyleDefaults(style: SplineMaterialStyle): {
  roughness: number;
  metalness: number;
  transmission: number;
  emissiveIntensity: number;
  clearcoat: number;
  ior: number;
} {
  switch (style) {
    case 'glass':
      return { roughness: 0.08, metalness: 0.0, transmission: 0.9, emissiveIntensity: 0, clearcoat: 0.85, ior: 1.5 };
    case 'chrome':
      return { roughness: 0.08, metalness: 1.0, transmission: 0, emissiveIntensity: 0, clearcoat: 0.5, ior: 1.5 };
    case 'clay':
      return { roughness: 0.9, metalness: 0.0, transmission: 0, emissiveIntensity: 0, clearcoat: 0, ior: 1.4 };
    case 'ceramic':
      return { roughness: 0.16, metalness: 0.02, transmission: 0, emissiveIntensity: 0, clearcoat: 1.0, ior: 1.5 };
    case 'neon':
      return { roughness: 0.25, metalness: 0.1, transmission: 0, emissiveIntensity: 1.6, clearcoat: 0.2, ior: 1.45 };
    case 'physical':
    default:
      return { roughness: 0.35, metalness: 0.12, transmission: 0, emissiveIntensity: 0, clearcoat: 0.25, ior: 1.45 };
  }
}

export function applySplineStyleToPhysicalMaterial(
  mat: THREE.MeshPhysicalMaterial,
  style: SplineMaterialStyle,
  hexColor: string,
  overrides?: Partial<{
    roughness: number;
    metalness: number;
    transmission: number;
    emissiveIntensity: number;
  }>
) {
  const defs = getStyleDefaults(style);
  const color = new THREE.Color(hexColor);
  mat.color.copy(color);
  mat.roughness = overrides?.roughness ?? defs.roughness;
  mat.metalness = overrides?.metalness ?? defs.metalness;
  mat.transmission = overrides?.transmission ?? defs.transmission;
  mat.clearcoat = defs.clearcoat;
  mat.ior = defs.ior;
  mat.thickness = mat.transmission > 0.02 ? 0.85 : 0;
  mat.transparent = mat.transmission > 0.02;
  mat.opacity = 1;
  const emissiveVal = overrides?.emissiveIntensity ?? defs.emissiveIntensity;
  if (emissiveVal > 0.01) {
    mat.emissive.copy(color);
    mat.emissiveIntensity = emissiveVal;
  } else {
    mat.emissive.setHex(0x000000);
    mat.emissiveIntensity = 0;
  }
  mat.needsUpdate = true;
}

function createGradientDomeTexture(topHex: string, horizonHex: string, bottomHex: string): THREE.CanvasTexture {
  const canvas = document.createElement('canvas');
  canvas.width = 16;
  canvas.height = 256;
  const ctx = canvas.getContext('2d')!;
  const grad = ctx.createLinearGradient(0, 0, 0, 256);
  grad.addColorStop(0, topHex);
  grad.addColorStop(0.5, horizonHex);
  grad.addColorStop(0.56, bottomHex);
  grad.addColorStop(1, bottomHex);
  ctx.fillStyle = grad;
  ctx.fillRect(0, 0, 16, 256);
  const tex = new THREE.CanvasTexture(canvas);
  tex.colorSpace = THREE.SRGBColorSpace;
  tex.needsUpdate = true;
  return tex;
}

        { color: 0xf1f5f9, intensity: 15, pos: [5, 11, 0], scale: [1.5, 10, 0.2], rotX: Math.PI / 2 },
        { color: 0xfbbf24, intensity: 12, pos: [0, 6, -10], scale: [3.5, 3.5, 0.2] },
        { color: 0x94a3b8, intensity: 5, pos: [10, 4, 6], scale: [5, 5, 0.2], rotY: -Math.PI / 2 },
      ],
    },
  };

  const spec = presets[id];
  if (!spec) return null;

  const envScene = new THREE.Scene();
  const domeGeo = new THREE.SphereGeometry(28, 24, 16);
  const domeTex = createGradientDomeTexture(spec.top, spec.horizon, spec.bottom);
  const domeMat = new THREE.MeshBasicMaterial({ map: domeTex, side: THREE.BackSide });
  envScene.add(new THREE.Mesh(domeGeo, domeMat));

  const boxGeo = new THREE.BoxGeometry(1, 1, 1);
  const createdMats: THREE.Material[] = [domeMat];
  for (const p of spec.panels) {
    const mat = new THREE.MeshLambertMaterial({
      color: 0x000000,
      emissive: p.color,
      emissiveIntensity: p.intensity,
    });
    createdMats.push(mat);
    const mesh = new THREE.Mesh(boxGeo, mat);
    mesh.position.set(...p.pos);
    mesh.scale.set(...p.scale);
    if (p.rotY) mesh.rotation.y = p.rotY;
    if (p.rotX) mesh.rotation.x = p.rotX;
    envScene.add(mesh);
  }

  const tex = pmremGenerator.fromScene(envScene, 0.04).texture;
  domeGeo.dispose();
  boxGeo.dispose();
  domeTex.dispose();
  createdMats.forEach((m) => m.dispose());
  return tex;
}

export function buildThreeLightFromSpec(
  spec: StudioLightSpec,
  castShadow: boolean
): THREE.Light {
  let light: THREE.Light;
  if (spec.type === 'dir') {
    light = new THREE.DirectionalLight(spec.color, spec.intensity);
  } else if (spec.type === 'point') {
    light = new THREE.PointLight(spec.color, spec.intensity);
  } else if (spec.type === 'spot') {
    light = new THREE.SpotLight(spec.color, spec.intensity, 0, 0.52, 0.4);
  } else {
    light = new THREE.AmbientLight(spec.color, spec.intensity);
  }

  light.name = spec.name;
  if (spec.type !== 'amb') {
    light.position.set(...spec.position);
    light.castShadow = castShadow;
    const shadowLight = light as THREE.DirectionalLight | THREE.PointLight | THREE.SpotLight;
    if (shadowLight.shadow) {
      shadowLight.shadow.mapSize.set(1024, 1024);
      shadowLight.shadow.bias = -0.0005;
    }
  }
  return light;
}

export function renderSceneToPngBlob(options: {
  renderer: THREE.WebGLRenderer;
  scene: THREE.Scene;
  camera: THREE.PerspectiveCamera;
  container: HTMLElement;
  resolution: RenderResolutionPreset;
  transparentBg: boolean;
  hideObjects?: THREE.Object3D[];
}): Promise<Blob | null> {
  const {
    renderer,
    scene,
    camera,
    container,
    resolution,
    transparentBg,
    hideObjects = [],
  } = options;

  const cw = Math.max(1, container.clientWidth);
  const ch = Math.max(1, container.clientHeight);
  const [targetW, targetH] =
    resolution === 'view' ? [cw, ch] : resolution.split('x').map(Number);

  const prevPixelRatio = renderer.getPixelRatio();
  const prevAspect = camera.aspect;
  const prevBg = scene.background;
  const prevFog = scene.fog;
  const prevClearAlpha = renderer.getClearAlpha();
  const prevVis = hideObjects.map((o) => o.visible);

  hideObjects.forEach((o) => {
    o.visible = false;
  });

  if (transparentBg) {
    scene.background = null;
    scene.fog = null;
    renderer.setClearAlpha(0);
  }

  renderer.setPixelRatio(1);
  renderer.setSize(targetW, targetH, false);
  camera.aspect = targetW / targetH;
  camera.updateProjectionMatrix();
  renderer.render(scene, camera);

  return new Promise((resolve) => {
    renderer.domElement.toBlob((blob) => {
      renderer.setPixelRatio(prevPixelRatio);
      renderer.setSize(cw, ch, false);
      camera.aspect = prevAspect;
      camera.updateProjectionMatrix();
      scene.background = prevBg;
      scene.fog = prevFog;
      renderer.setClearAlpha(prevClearAlpha);
      hideObjects.forEach((o, idx) => {
        o.visible = prevVis[idx];
      });
      renderer.render(scene, camera);
      resolve(blob);
    }, 'image/png');
  });
}
